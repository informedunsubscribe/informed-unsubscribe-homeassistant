#include <tunables/global>

# Home Assistant loads this for the app and renames the profile to its own id for it (such as
# "266e2efd_informed_unsubscribe"), so rules refer to the profile as @{profile_name}.
# To change it, add "complain," to the flags (refusals are then only logged), use every feature
# on Home Assistant, and check `journalctl -t audit` on the host for apparmor="ALLOWED" lines.
profile informed_unsubscribe flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Home Assistant stops the app with a signal; the app stops its own helpers on a timeout.
  signal (receive) peer=unconfined,
  signal (send,receive) peer=@{profile_name},

  # IMAP, SMTP, the sender lookups, the ingress web page and Home Assistant's notification API.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # The only programs it runs: Python (the app and its per-run helper) and Tesseract.
  /usr/bin/python3* ix,
  /usr/bin/tesseract ix,

  /opt/informed_unsubscribe/ r,
  /opt/informed_unsubscribe/** r,
  /lib/** mr,
  /usr/lib/** mr,
  /usr/share/tessdata/ r,
  /usr/share/tessdata/** r,
  /usr/share/zoneinfo/** r,
  /etc/ssl/** r,
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/localtime r,

  @{PROC}/cpuinfo r,
  @{PROC}/@{pid}/fd/ r,
  /dev/urandom r,
  /dev/null rw,

  # Options, the database and the scans. Nothing else is written.
  /data/ r,
  /data/** rwk,
}
